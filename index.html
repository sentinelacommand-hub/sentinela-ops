rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ========================================== // FUNÇÕES AUXILIARES // ========================================== function logado() { return request.auth != null; } function emailAdminDireto() { return request.auth.token.email == 'admin@sentinela.com' || request.auth.token.email == 'gerencia@sentinela.com'; } function roleAdminNoBanco() { return exists(/databases/$(database)/documents/usuarios/$(request.auth.uid)) && (get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.get('role', '') == 'admin' || get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.get('role', '') == 'gerencia'); } function ehAdmin() { return logado() && (emailAdminDireto() || roleAdminNoBanco()); } // ========================================== // REGRAS DAS COLEÇÕES // ========================================== match /usuarios/{userId} { allow read, write: if logado() && request.auth.uid == userId; allow read, write: if ehAdmin(); } match /postos/{postoId} { allow read: if logado(); allow write: if ehAdmin(); } match /rondas/{rondaId} { // CORRIGIDO: Agora permite que o Admin crie rondas para outros UIDs allow create: if logado() && (request.resource.data.uid == request.auth.uid || ehAdmin()); allow read: if logado() && (request.auth.uid == resource.data.uid || ehAdmin()); allow update, delete: if ehAdmin(); } match /solicitacoes/{solicitacaoId} { allow create: if logado(); allow read: if logado() && (request.auth.uid == resource.data.uid || ehAdmin()); allow update, delete: if ehAdmin(); } match /visitantes_heatmap/{documentId} { allow read: if logado(); allow create, update, delete: if ehAdmin(); } } }
